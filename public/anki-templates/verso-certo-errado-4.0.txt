<script>
function alternarComentarioProfessor() {
  var botao = document.getElementById('lf_professor_toggle');
  var conteudo = document.getElementById('lf_professor_content');
  var icone = document.getElementById('lf_professor_icon');
  if (!botao || !conteudo || !icone) return;

  var abrir = conteudo.hasAttribute('hidden');
  if (abrir) {
    conteudo.removeAttribute('hidden');
  } else {
    conteudo.setAttribute('hidden', 'hidden');
  }

  botao.setAttribute('aria-expanded', abrir ? 'true' : 'false');
  icone.innerHTML = abrir ? '&#9660;' : '&#9654;';
}
</script>

<script>
(function () {
  var total = Number('{{TotalArtigos}}'.replace(/[^0-9.]/g, '')) || 0;
  var ref = String('{{ordem}}') || '';
  var idx = 0;
  var mOrdem = ref.match(/^\s*(\d{4})/);

  if (mOrdem) idx = parseInt(mOrdem[1], 10) || 0;

  var pctFinal = (total > 0 && idx > 0)
    ? Math.max(0, Math.min(100, Math.round((idx / total) * 100)))
    : 0;

  var bar = document.getElementById('lf_pb');
  var pctSpan = document.getElementById('lf_pct');
  var textoBarra = document.querySelector('.lf-progress-label');
  var progresso = document.querySelector('.lf-progress-row');

  if (bar && pctSpan) {
    bar.style.width = pctFinal + '%';
    pctSpan.textContent = pctFinal + '%';
  }

  if (pctFinal === 100) {
    if (textoBarra) textoBarra.textContent = 'Lei concluída.';
    if (progresso) progresso.classList.add('lf-progress-complete');
  }

  var escolha = window.localStorage.getItem('escolha_usuario') || '';
  var resposta = '{{resposta}}';

  function limpar(txt) {
    return (txt || '')
      .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
      .replace(/\s+/g, ' ')
      .trim().toLowerCase();
  }

  var correcaoEl = document.getElementById('correcao');
  var painelResultado = document.getElementById('lf_result_panel');
  var recompensaEl = document.getElementById('lf_reward');
  var botaoCerto = document.getElementById('lf_back_certo');
  var botaoErrado = document.getElementById('lf_back_errado');
  if (!correcaoEl) return;

  if (limpar(escolha) === 'certo' && botaoCerto) {
    botaoCerto.classList.add('selecionado');
  } else if (limpar(escolha) === 'errado' && botaoErrado) {
    botaoErrado.classList.add('selecionado');
  }

  if (!escolha) {
    correcaoEl.innerHTML = '<span class="lf-result-neutral">Nenhuma resposta foi marcada na frente.</span>';
    return;
  }

  if (limpar(escolha) === limpar(resposta)) {
    correcaoEl.innerHTML = '<span class="lf-result-icon">&#10003;</span><span><strong>Parabéns! Você acertou.</strong></span>';
    if (painelResultado) painelResultado.classList.add('lf-result-correct');
  } else {
    correcaoEl.innerHTML = '<span class="lf-result-icon">&#10005;</span><span><strong>Você errou.</strong></span>';
    if (painelResultado) painelResultado.classList.add('lf-result-wrong');

    if (recompensaEl) {
      recompensaEl.textContent = 'Revise e tente novamente.';
      recompensaEl.classList.add('mostrar');
    }
  }
})();
</script>
<script>
(function () {
  var botoesLegisbot = document.querySelectorAll('.lf-card-actions .lf-legisbot-link');
  if (!botoesLegisbot.length) return;

  function valorDoCampo(id, preservarTexto) {
    var elemento = document.getElementById(id);
    if (!elemento) return '';
    var valor = typeof elemento.value === 'string'
      ? elemento.value
      : (elemento.textContent || '');
    return preservarTexto ? valor : valor.trim();
  }

  var slug = '{{slug}}'.trim();
  var ordem = '{{ordem}}'.trim();
  var titulo = valorDoCampo('lf_legisbot_titulo', false);
  var assunto = valorDoCampo('lf_legisbot_assunto', false);
  var legislacao = valorDoCampo('lf_legisbot_trecho', true);
  var base = 'https://www.legisflashcards.com.br/legisbot/';
  var hrefBase = base
    + encodeURIComponent(slug)
    + '/'
    + encodeURIComponent(ordem)
    + '?titulo='
    + encodeURIComponent(titulo)
    + '&assunto='
    + encodeURIComponent(assunto)
    + '&legislacao='
    + encodeURIComponent(legislacao);

  for (var i = 0; i < botoesLegisbot.length; i++) {
    var aba = botoesLegisbot[i].getAttribute('data-aba') || 'legisbot';
    botoesLegisbot[i].href = hrefBase + '&aba=' + encodeURIComponent(aba);
  }
})();
</script>
