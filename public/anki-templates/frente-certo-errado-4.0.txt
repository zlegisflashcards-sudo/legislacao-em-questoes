<script>
(function () {
  document.querySelectorAll('table').forEach(function (tabela) {
    var parent = tabela.parentElement;
    if (!parent || parent.classList.contains('tabela-scroll')) return;
    var wrapper = document.createElement('div');
    wrapper.className = 'tabela-scroll';
    tabela.parentNode.insertBefore(wrapper, tabela);
    wrapper.appendChild(tabela);
  });
})();

(function () {
  if (!document.querySelector('.nightMode')) return;
  document.querySelectorAll('.lf-statement, .lf-statement *').forEach(function (el) {
    if (!el.style) return;
    el.style.removeProperty('background');
    el.style.removeProperty('background-color');
    el.style.removeProperty('background-image');
    el.style.removeProperty('color');
    el.style.removeProperty('filter');
    el.style.removeProperty('opacity');
  });
})();
</script>

<script>
function responder(valor) {
  window.localStorage.setItem('escolha_usuario', valor);

  var certo = document.getElementById('btnCerto');
  var errado = document.getElementById('btnErrado');
  certo.classList.remove('selecionado');
  errado.classList.remove('selecionado');

  if (valor === 'Certo') {
    certo.classList.add('selecionado');
  } else {
    errado.classList.add('selecionado');
  }
}
</script>

<script>
(function () {
  var link = document.getElementById('lf_law_update');
  if (!link) return;

  var slug = (link.getAttribute('data-slug') || '').trim();
  slug = slug.replace(/^\/+|\/+$/g, '');
  var siteBase = 'https://www.legisflashcards.com.br';

  link.href = (slug && slug.charAt(0) !== '{')
    ? siteBase + '/leisflashcards/' + encodeURIComponent(slug)
    : siteBase;
})();
</script>

<script>
(function () {
  var link = document.getElementById('lf_report_error');
  if (!link) return;
  var ordem = link.getAttribute('data-ordem') || '';
  var tituloEl = document.getElementById('lf_report_titulo');
  var assuntoEl = document.getElementById('lf_report_assunto');
  var titulo = tituloEl ? tituloEl.textContent.trim() : '';
  var assuntoCard = assuntoEl ? assuntoEl.textContent.trim() : '';
  var assuntoEmail = 'ERRO NO FLASHCARD - ' + (assuntoCard || 'Assunto');
  var corpo = '\uD83D\uDCDD Descrição do erro:\n(Descreva aqui)\n\n\uD83D\uDCF7 Se possível, envie um print em anexo.';
  link.href = 'mailto:zlegisflashcards@gmail.com?subject=' + encodeURIComponent(assuntoEmail)
    + '&body=' + encodeURIComponent(corpo);
})();
</script>

<script>
(function () {
  var total = Number('{{TotalArtigos}}'.replace(/[^0-9.]/g, '')) || 0;
  var ref = String('{{ordem}}') || '';
  var idx = 0;
  var mOrdem = ref.match(/^\s*(\d{4})/);

  if (mOrdem) idx = parseInt(mOrdem[1], 10) || 0;

  var pctFinal = (total > 0 && idx > 0)
    ? Math.max(0, Math.min(100, Math.round((idx / total) * 100)))
    : 0;

  var bar = document.getElementById('lf_pb');
  var pctSpan = document.getElementById('lf_pct');
  var textoBarra = document.querySelector('.lf-progress-label');
  var progresso = document.querySelector('.lf-progress-row');

  if (bar && pctSpan) {
    bar.style.width = pctFinal + '%';
    pctSpan.textContent = pctFinal + '%';
  }

  if (pctFinal === 100) {
    if (textoBarra) textoBarra.textContent = 'Lei concluída.';
    if (progresso) progresso.classList.add('lf-progress-complete');
  }
})();
</script>
